euclid-cdk

Signed package repositories for the C++ SDK for a euclid server.

curl -fsSL https://jensvogt.github.io/euclid-cdk/euclid-cdk.gpg.asc | sudo gpg --dearmor -o /usr/share/keyrings/euclid-cdk.gpg
echo "deb [signed-by=/usr/share/keyrings/euclid-cdk.gpg] https://jensvogt.github.io/euclid-cdk/apt stable main" | sudo tee /etc/apt/sources.list.d/euclid-cdk.list
sudo apt update && sudo apt install euclid-cdk
sudo curl -fsSL https://jensvogt.github.io/euclid-cdk/euclid-cdk.repo -o /etc/yum.repos.d/euclid-cdk.repo
sudo yum install euclid-cdk
